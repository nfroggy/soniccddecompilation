function(soniccd_add_variant variant)
  string(TOLOWER "${variant}" target_name)
  set(target "${target_name}")

  set(target_sources ${${variant}_SOURCES})
  if(NOT target_sources)
    message(FATAL_ERROR "No source list defined for variant ${variant}")
  endif()

  add_library(${target} SHARED ${target_sources})
  set_target_properties(${target} PROPERTIES
    WINDOWS_EXPORT_ALL_SYMBOLS ON
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin
    LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin
    ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib
  )
  
  set_target_properties(${target} PROPERTIES
    C_STANDARD 17
    C_STANDARD_REQUIRED ON
  )

  target_include_directories(${target} PRIVATE ${SONICCD_INCLUDE_DIRS})
  target_compile_definitions(${target} PRIVATE ${${variant}_DEFINES})
  source_group(TREE ${CMAKE_CURRENT_SOURCE_DIR} FILES ${target_sources})

  if(MSVC)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS WIN32 _WINDOWS)
    target_compile_options(${target} PRIVATE
      /W3
      "/FI${SONICCD_MSVC_COMPAT_HEADER}"
    )
  else()
    target_compile_options(${target} PRIVATE
      -Wall
    )
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
      target_compile_options(${target} PRIVATE -fcommon)
    endif()
  endif()
endfunction()

if(SONICCD_VARIANTS STREQUAL "all")
  set(SONICCD_SELECTED_VARIANTS ${SONICCD_ALL_VARIANTS})
else()
  set(SONICCD_SELECTED_VARIANTS ${SONICCD_VARIANTS})
endif()

foreach(variant IN LISTS SONICCD_SELECTED_VARIANTS)
  list(FIND SONICCD_ALL_VARIANTS "${variant}" variant_index)
  if(variant_index EQUAL -1)
    message(FATAL_ERROR "Unknown SONICCD variant: ${variant}")
  endif()
  soniccd_add_variant(${variant})
  string(TOLOWER "${variant}" target_name)
  list(APPEND SONICCD_SELECTED_TARGETS ${target_name})
endforeach()

add_custom_target(variants DEPENDS ${SONICCD_SELECTED_TARGETS})

# use vendored sdl3 lib on msvc

  target_include_directories(frontend PRIVATE "libs/SDL3/include")
  target_include_directories(frontend PRIVATE "libs/SDL3_mixer/include")
  if(CMAKE_CL_64)
	target_link_libraries(frontend PRIVATE "${PROJECT_SOURCE_DIR}/libs/SDL3/lib/x64/SDL3.lib")
	target_link_libraries(frontend PRIVATE "${PROJECT_SOURCE_DIR}/libs/SDL3_mixer/lib/x64/SDL3_mixer.lib")
	set(MSVC_PLATFORM_DLL_PATH ${PROJECT_SOURCE_DIR}/libs/SDL3/lib/x64;${PROJECT_SOURCE_DIR}/libs/SDL3_mixer/lib/x64)
  else()
    target_link_libraries(frontend PRIVATE "${PROJECT_SOURCE_DIR}/libs/SDL3/lib/x86/SDL3.lib")
	target_link_libraries(frontend PRIVATE "${PROJECT_SOURCE_DIR}/libs/SDL3_mixer/lib/x86/SDL3_mixer.lib")
	set(MSVC_PLATFORM_DLL_PATH ${PROJECT_SOURCE_DIR}/libs/SDL3/lib/x86;${PROJECT_SOURCE_DIR}/libs/SDL3_mixer/lib/x86)
  endif()
  # make sure the debugger can find the sdl dll file
  set(FRONTEND_PATH "PATH=%PATH%" ${MSVC_PLATFORM_DLL_PATH})
  set_target_properties(frontend PROPERTIES VS_DEBUGGER_ENVIRONMENT "${FRONTEND_PATH}")
  # set the debugger working directory
  set_target_properties(frontend PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
# otherwise use system sdl3
else()
  find_package(SDL3 REQUIRED CONFIG REQUIRED COMPONENTS SDL3)
  target_link_libraries(frontend PRIVATE SDL3::SDL3)
endif()
